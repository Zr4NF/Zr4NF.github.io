# 反编译
```c
int __fastcall __noreturn main(int argc, const char **argv, const char **envp)
{
  int i; // [rsp+8h] [rbp-118h]
  int v4; // [rsp+Ch] [rbp-114h]
  char buf[264]; // [rsp+10h] [rbp-110h] BYREF
  unsigned __int64 v6; // [rsp+118h] [rbp-8h]

  v6 = __readfsqword(0x28u);
  init();
  puts("呐~不会吧不会吧？不会真有人连这么简单的flag都拿不到吧？诶嘿~♡");
  puts("杂~鱼~就乖乖承认自己是笨蛋好啦！略略略~");
  puts("♡~~~~~~~~~~~~~~~~~~~~~~~~~~~~♡");
  system("ls -l");
  puts("♡~~~~~~~~~~~~~~~~~~~~~~~~~~~~♡");
  v4 = read(0, buf, 0x100u);
  buf[v4] = 0;
  for ( i = 0; i < v4; ++i )
  {
    if ( (buf[i] <= '/' || buf[i] > '9')
      && (buf[i] <= '@' || buf[i] > 'Z')
      && (buf[i] <= '`' || buf[i] > 'z')
      && buf[i] != ' '
      && buf[i] != '\n'
      && buf[i] != '.'
      && buf[i] != '_' )
    {
      puts("啊啦~杂鱼用了不允许的字符呢~想骗过人家？不行哦~");
      exit(0);
    }
  }
  if ( strncmp(buf, "cat", 3u) )
  {
    puts("哼~命令必须以cat开头哦~杂鱼连这都不知道吗？");
    exit(0);
  }
  if ( strstr(buf, "flag") )
  {
    puts("啊啦~想偷偷看flag？不行不行~杂鱼没有权限哦~");
    exit(0);
  }
  system(buf);
  puts("♡~~~~~~~~~~~~~~~~~~~~~~~~~~~~♡");
  puts("杂鱼~拿不到flag的杂鱼~");
  close(1);
  close(2);
  buf[(int)read(0, buf, 0x100u)] = 0;
  system(buf);
  close(0);
  exit(0);
}
```

可以看到第二次没有做字符过滤，但是关闭了 fd1 和 fd 2，可以通过`cat flag 1>&0`或`cat falg >&0`获得flag。

*tips：追加管道符`>>`，不适用于`&0`写法，shell会报错*
